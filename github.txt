<script>
const statuses=["To Do","In Progress","Review","Done"];
const initial=[
{id:1,title:"Finalize event requirements",assignee:"Aarav",due:"2026-11-02",status:"To Do"},
{id:2,title:"Prepare event budget",assignee:"Priya",due:"2026-11-04",status:"To Do"},
{id:3,title:"Contact sponsors",assignee:"Rahul",due:"2026-11-06",status:"To Do"},
{id:4,title:"Design event posters",assignee:"Sneha",due:"2026-11-01",status:"In Progress"},
{id:5,title:"Review registration form",assignee:"Priya",due:"2026-10-31",status:"Review"},
{id:6,title:"Create event planning checklist",assignee:"Aarav",due:"2026-10-28",status:"Done"}
];
let tasks;
try{tasks=JSON.parse(localStorage.getItem("csmpTasks"))||initial}
catch(e){tasks=initial}
let nextId=Math.max(6,...tasks.map(t=>t.id))+1;
function save(){localStorage.setItem("csmpTasks",JSON.stringify(tasks))}
function render(){
const board=document.getElementById("board");
board.innerHTML="";
statuses.forEach((status,index)=>{
const items=tasks.filter(t=>t.status===status);
const col=document.createElement("section");
col.className="column";
col.ondragover=e=>e.preventDefault();
col.ondrop=e=>{
e.preventDefault();
const id=Number(e.dataTransfer.getData("text/plain"));
const task=tasks.find(t=>t.id===id);
if(task){task.status=status;save();render()}
};
const heading=document.createElement("h3");
heading.textContent=status+" ";
const count=document.createElement("span");
count.className="count";count.textContent=items.length;
heading.append(count);col.append(heading);
items.forEach(task=>{
const card=document.createElement("article");
card.className="task";card.draggable=true;
card.ondragstart=e=>e.dataTransfer.setData("text/plain",task.id);
const id=document.createElement("div");id.className="id";
id.textContent="EM-"+String(task.id).padStart(2,"0");
const title=document.createElement("h4");title.textContent=task.title;
const person=document.createElement("div");person.className="meta";
person.textContent="Assignee: "+task.assignee;
const date=document.createElement("div");date.className="meta";
date.textContent="Due: "+task.due;
const move=document.createElement("select");
statuses.forEach(s=>{const o=document.createElement("option");o.textContent=s;o.value=s;move.append(o)});
move.value=task.status;
move.setAttribute("aria-label","Move "+task.title);
move.onchange=()=>{task.status=move.value;save();render()};
card.append(id,title,person,date,move);col.append(card);
});
board.append(col);
});
document.getElementById("total").textContent=tasks.length;
["To Do","In Progress","Review","Done"].forEach((s,i)=>{
document.getElementById(["todo","progress","review","done"][i]).textContent=
tasks.filter(t=>t.status===s).length;
});
}
document.getElementById("taskform").onsubmit=e=>{
e.preventDefault();
tasks.push({
id:nextId++,
title:document.getElementById("title").value.trim(),
assignee:document.getElementById("assignee").value.trim(),
due:document.getElementById("due").value,
status:document.getElementById("status").value
});
save();render();e.target.reset();e.target.hidden=true;
};
render();
</script>
